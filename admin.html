// Gentle Storm Racing (GSR) - Core API Worker Proxy v2.0 // Serves: CMS Draft/Live Pipeline, Store Manager, Moderation Queue & SL Telemetry const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, x-race-secret, x-sl-key", }; export default { async fetch(request, env, ctx) { // Handle CORS Preflight Options if (request.method === "OPTIONS") { return new Response(null, { headers: CORS_HEADERS }); } const url = new URL(request.url); const path = url.pathname; try { // ---------------------------------------------------------------------- // PUBLIC ENDPOINTS // ---------------------------------------------------------------------- // 1. GET /store/items - Fetch published store items if (path === "/store/items" && request.method === "GET") { const queryUrl = `${env.SUPABASE_URL}/rest/v1/store_items?status=eq.published&select=*&order=created_at.desc`; return await supabaseFetch(queryUrl, "GET", null, env); } // 2. GET /cms/live?page=main_menu - Fetch live CMS content for public pages if (path === "/cms/live" && request.method === "GET") { const pageName = url.searchParams.get("page") || "main_menu"; const queryUrl = `${env.SUPABASE_URL}/rest/v1/page_content?page_name=eq.${pageName}&select=section_order,layout_type,live_data&order=section_order.asc`; return await supabaseFetch(queryUrl, "GET", null, env); } // ---------------------------------------------------------------------- // ADMIN ENDPOINTS (Requires x-race-secret Header Authorization) // ---------------------------------------------------------------------- const adminSecret = request.headers.get("x-race-secret"); if (path.startsWith("/admin") && adminSecret !== env.ADMIN_RACE_SECRET) { return new Response(JSON.stringify({ error: "Unauthorized access" }), { status: 401, headers: { ...CORS_HEADERS, "Content-Type": "application/json" }, }); } // 3. POST /admin/store/save - Publish/Update store item if (path === "/admin/store/save" && request.method === "POST") { const payload = await request.json(); const queryUrl = `${env.SUPABASE_URL}/rest/v1/store_items`; return await supabaseFetch(queryUrl, "POST", payload, env, { "Prefer": "resolution=merge-duplicates" }); } // 4. GET /admin/cms/draft?page=main_menu - Load draft CMS content for iPad editor if (path === "/admin/cms/draft" && request.method === "GET") { const pageName = url.searchParams.get("page") || "main_menu"; const queryUrl = `${env.SUPABASE_URL}/rest/v1/page_content?page_name=eq.${pageName}&select=*&order=section_order.asc`; return await supabaseFetch(queryUrl, "GET", null, env); } // 5. POST /admin/cms/save-draft - Save draft changes from iPad editor if (path === "/admin/cms/save-draft" && request.method === "POST") { const payload = await request.json(); // Array of section objects const queryUrl = `${env.SUPABASE_URL}/rest/v1/page_content`; return await supabaseFetch(queryUrl, "POST", payload, env, { "Prefer": "resolution=merge-duplicates" }); } // 6. POST /admin/cms/publish - Push draft_data to live_data for a specific page if (path === "/admin/cms/publish" && request.method === "POST") { const { page_name } = await request.json(); if (!page_name) throw new Error("Missing page_name parameter"); // Execute SQL RPC function via Supabase to copy draft_data -> live_data const rpcUrl = `${env.SUPABASE_URL}/rest/v1/rpc/publish_page_draft`; return await supabaseFetch(rpcUrl, "POST", { target_page: page_name }, env); } // 7. GET /admin/approvals - Fetch pending driver/team submissions if (path === "/admin/approvals" && request.method === "GET") { const queryUrl = `${env.SUPABASE_URL}/rest/v1/approval_queue?status=eq.pending&select=*&order=created_at.asc`; return await supabaseFetch(queryUrl, "GET", null, env); } // 8. POST /admin/approvals/review - Approve or Reject pending submission if (path === "/admin/approvals/review" && request.method === "POST") { const { approval_id, action, reviewer_key } = await request.json(); const rpcUrl = `${env.SUPABASE_URL}/rest/v1/rpc/process_approval`; return await supabaseFetch(rpcUrl, "POST", { approval_id, action, reviewer_key }, env); } return new Response(JSON.stringify({ error: "Endpoint not found" }), { status: 404, headers: { ...CORS_HEADERS, "Content-Type": "application/json" }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { ...CORS_HEADERS, "Content-Type": "application/json" }, }); } } }; // Helper function to query Supabase REST API using Service Role Key async function supabaseFetch(url, method, body, env, extraHeaders = {}) { const headers = { "apikey": env.SUPABASE_SERVICE_ROLE_KEY, "Authorization": `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`, "Content-Type": "application/json", ...extraHeaders }; const options = { method, headers }; if (body) options.body = JSON.stringify(body); const res = await fetch(url, options); const data = await res.text(); return new Response(data, { status: res.status, headers: { ...CORS_HEADERS, "Content-Type": "application/json" }, }); }
